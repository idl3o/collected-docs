    const [currentPath, setCurrentPath] = useState('');
    const [healingPaths, setHealingPaths] = useState([]);
    const [affirmation, setAffirmation] = useState('');
    const [biometricData, setBiometricData] = useState({});

        "Your creative flow is activating your healing response",
        "I witness the courage in your artistic expression",
        "Your art is creating new neural pathways for wellness",
        "Trust the wisdom flowing through your fingers"
    ];
    
    const handleTouchStart = (event) => {
        const { locationX, locationY } = event.nativeEvent;
        setCurrentPath(`M${locationX},${locationY}`);
        provideLiveEncouragement();
    };
    
    const handleTouchMove = (event) => {
        const { locationX, locationY } = event.nativeEvent;
        setCurrentPath(prev => `${prev} L${locationX},${locationY}`);
    };
    
    const handleTouchEnd = () => {
        setHealingPaths(prev => [...prev, currentPath]);
        setCurrentPath('');
        provideAffirmingFeedback();
    };
    
    const provideLiveEncouragement = () => {
        const encouragement = affirmingFeedback[Math.floor(Math.random() * affirmingFeedback.length)];

                <Text style={styles.biometricText}>💗 Heart: Coherent</Text>
